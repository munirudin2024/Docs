
dotnet new console -n NamaProjek
# : Membuat projek Console baru.

dotnet run
# : Menjalankan aplikasi.

dotnet build
# : Mengompilasi kode tanpa menjalankan.

dotnet watch
# : Menjalankan aplikasi dan otomatis restart kalau ada perubahan kode (sangat berguna!).

dotnet add package [NamaPackage]
# : Menambah library (mirip cargo add di Rust).
